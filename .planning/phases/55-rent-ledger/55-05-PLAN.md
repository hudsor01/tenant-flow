// rent-ledger-keys.ts
export type LedgerSummary = { chargesTotal: number; creditsTotal: number; receiptsTotal: number; balance: number; lateCount: number; lateAmount: number };
export type LedgerEntry = { id: string; kind: 'charge'|'receipt'; type: string; amount: number; entryDate: string; dueDate: string|null; reversesId: string|null; chargeId: string|null; receiptsSum: number };
export function mapLedgerSummaryRow(raw: Record<string, unknown>): LedgerSummary;
export function mapLedgerEntryRow(raw: Record<string, unknown>): LedgerEntry;
export const rentLedgerKeys: { all; forLease(leaseId): ... ; summary(leaseId); entries(leaseId) };
export const rentLedgerQueries: { summary(leaseId), entries(leaseId) };  // queryOptions()

// mutation inputs (dollars — NO cents)
recordReceipt: { chargeId: string; leaseId: string; amount: number; method: string; receivedDate: string }  // insert rent_receipts
addLine:       { leaseId: string; type: 'late_fee'|'manual_charge'|'credit'; amount: number; date: string; description: string }  // insert rent_charges (credit stored negative)
startTracking: { leaseId: string; startDate: string; openingBalance: number }  // rpc start_lease_ledger
reverseEntry:  { leaseId: string; entryKind: 'charge'|'receipt'; entryId: string; amount: number }  // charge -> rpc reverse_charge; receipt -> rpc reverse_receipt (atomic exact negation, W2)

// use-rent-ledger.ts hooks
useLedgerSummary(leaseId), useLedgerEntries(leaseId),
useRecordReceiptMutation(leaseId), useAddLineMutation(leaseId), useStartTrackingMutation(leaseId), useReverseEntryMutation(leaseId)
